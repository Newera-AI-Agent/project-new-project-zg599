    31	/** Expand IPv6 `::` shorthand to full 8-group form. */
    32	function expandIPv6(ip: string): string {
    33	    if (!ip.includes("::")) {
    34	        return ip
    35	            .split(":")
    36	            .map((group) => group.padStart(4, "0"))
    37	            .join(":");
    38	    }
    39	    const halves = ip.split("::");
    40	    const left = halves[0] ? halves[0].split(":") : [];
    41	    const right = halves[1] ? halves[1].split(":") : [];
    42	    const middle = Array(8 - left.length - right.length).fill("0000");
    43	    return [...left, ...middle, ...right]
    44	        .map((group) => group.padStart(4, "0"))
    45	        .join(":");
    46	}

    48	/** Truncate IP to /24 subnet (IPv4) or /48 subnet (IPv6, first 3 groups). */
    49	export function truncateIpToSubnet(ip: string | undefined): string | undefined {
    50	    if (!ip) return undefined;
    51	    const normalized = stripIPv4MappedPrefix(ip);
    52	    if (normalized.includes(".")) {
    53	        const parts = normalized.split(".");
    54	        if (parts.length === 4) return `${parts[0]}.${parts[1]}.${parts[2]}.0`;
    55	    }
    56	    if (normalized.includes(":")) {
    57	        const groups = expandIPv6(normalized).split(":");
    58	        return `${groups[0]}:${groups[1]}:${groups[2]}::`;
    59	    }
    60	    return undefined;

    15	/**
    16	 * Creates a cached version of an async function using Cloudflare KV
    17	 * @param fn The async function to cache
    18	 * @param options Cache configuration options
    19	 * @returns A cached version of the function
    20	 */
    21	export function cached<TArgs extends unknown[], TReturn>(
    22	    fn: (...args: TArgs) => Promise<TReturn>,
    23	    options: CacheOptions<TArgs>,
    24	): (...args: TArgs) => Promise<TReturn> {
    25	    const { ttl, kv, keyGenerator, log } = options;
    26	
    27	    return async (...args: TArgs): Promise<TReturn> => {
    28	        const cacheKey = await keyGenerator(...args);
    29	
    30	        try {
    31	            log.trace("Fetching from cache: {key}", { key: cacheKey });
    32	            const cachedData = (await kv.get(
    33	                cacheKey,
    34	                "json",
    35	            )) as CacheEntry<TReturn> | null;
    36	            if (cachedData) {
    37	                const ttlChanged = cachedData.ttl !== ttl;
    38	                if (!ttlChanged) {
    39	                    return cachedData.value;
    40	                }
    41	            }
    42	        } catch (error) {
    43	            log.warn("Failed to read from cache: {error}", { error });
    44	        }
    45	
    46	        log.trace("Executing function: {key}", { key: cacheKey });
    47	        const result = await fn(...args);
    48	
    49	        try {
    50	            log.trace("Writing to cache: {key}", { key: cacheKey });
    51	            const cacheEntry: CacheEntry<TReturn> = {
    52	                value: result,
    53	                ttl,
    54	            };
    55	            await kv.put(cacheKey, JSON.stringify(cacheEntry), {
    56	                expirationTtl: ttl,
    57	            });
    58	        } catch (error) {
    59	            log.warn("Failed to write to cache: {error}", { error });
    60	        }
    61	
    62	        return result;
    63	    };

     9	        EDGE_RATE_LIMITER?: RateLimit;
    10	        EDGE_IP_RATE_LIMITER?: RateLimit;
    11	    };
    12	};

    14	/**
    15	 * Cloudflare-native limit that runs before auth, so one flooding key is
    16	 * rejected without reaching D1. Counts per API key, hashed so no lookup is
    17	 * needed. With `byIp`, requests without a key count per client IP instead,
    18	 * and keyed requests also count per IP in `EDGE_IP_RATE_LIMITER`, so
    19	 * rotating made-up keys from one IP cannot dodge the limit. Each Worker's
    20	 * wrangler.toml sets its limits per 10 seconds, matching `Retry-After`.
    21	 * Enter leaves `byIp` off because gen's internal calls arrive without a
    22	 * client IP; it still counts every request that carries a client IP, keyed
    23	 * or not, so public routes such as device codes, app lookup and sign-in
    24	 * cannot be flooded into D1 with made-up keys. Cloudflare counts per
    25	 * location, so this caps floods rather than enforcing an exact quota.
    26	 */
    27	export function rateLimit({ byIp }: { byIp: boolean }) {
    28	    return createMiddleware<RateLimitEnv>(async (c, next) => {
    29	        const apiKey = extractApiKey(c.req.raw);
    30	        const ip = `ip:${getRealClientIp(c)}`;
    31	        if (apiKey) {
    32	            const digest = await crypto.subtle.digest(
    33	                "SHA-256",
    34	                new TextEncoder().encode(apiKey),
    35	            );
    36	            await limit(
    37	                c,
    38	                c.env.EDGE_RATE_LIMITER,
    39	                `key:${bytesToHex(digest)}`,
    40	            );
    41	            if (byIp) await limit(c, c.env.EDGE_IP_RATE_LIMITER, ip);
    42	        } else if (byIp) {
    43	            await limit(c, c.env.EDGE_RATE_LIMITER, ip);
    44	        }
    45	        if (!byIp && c.req.header("cf-connecting-ip")) {
    46	            await limit(c, c.env.EDGE_RATE_LIMITER, ip);
    47	        }
    48	        await next();
    49	    });
    50	}
    51	
    52	/** A 429 once `key` is over `limiter`; `retryAfter` is its period in seconds. */
    53	export async function limit(
    54	    c: Context,
    55	    limiter: RateLimit | undefined,
    56	    key: string,
    57	    retryAfter = 10,
    58	) {
    59	    if (limiter && !(await limiter.limit({ key })).success) {
    60	        c.header("Retry-After", String(retryAfter));
    61	        throw new HTTPException(429, {
    62	            message: "Too many requests. Please slow down.",
    63	        });
    64	    }

     1	/**
     2	 * Match an incoming `redirect_uri` against a registered allowlist for a `pk_`.
     3	 *
     4	 * Scheme policy (https-only except loopback http) is enforced at registration
     5	 * time (see validateRedirectUriFormat). Matching trusts the stored allowlist
     6	 * and does not re-check the scheme.
     7	 *
     8	 * Comparison rules:
     9	 * - Scheme + hostname + path are matched exactly (after lowercasing host).
    10	 * - Trailing slash on the path is insignificant: `/cb` and `/cb/` match.
    11	 * - Query strings are ignored when the registered URI has no query: apps
    12	 *   round-trip state (e.g. `?prompt=...`) and the auth code/token rides the URL
    13	 *   fragment, not the query. If the registered URI includes a query, it must
    14	 *   match exactly so existing query-bound allowlist entries are not broadened.
    15	 * - For loopback entries (RFC 8252 §7.3): port is ignored — any port matches.
    16	 *   This covers native/CLI apps that bind to ephemeral ports each run.
    17	 * - For non-loopback entries: port must match (default ports normalized).
    18	 * - Fragments are rejected; redirect URIs must not contain them.
    19	 *
    20	 * Returns false on any parse error or empty allowlist.
    21	 *
    22	 * Single source of truth for both /api/app-lookup (consent screen attribution)
    23	 * and createApiKeyForUser (sk_ minting). Keep them aligned.
    24	 */
    25	export function redirectUriMatchesAllowlist(
    26	    uri: string,
    27	    allowlist: readonly string[] | null | undefined,
    28	): boolean {
    29	    if (!allowlist?.length) return false;
    30	    const incoming = safeParse(uri);
    31	    if (!incoming) return false;
    32	    return allowlist.some((entry) => matchesEntry(incoming, entry));
    33	}
    34	
    35	/**
    36	 * Strict variant for the OAuth authorization-code flow (RFC 6749 §3.1.2.3):
    37	 * the authorization code rides the redirect query string, so the fragment-era
    38	 * leniency of accepting extra query params must not apply. The query must
    39	 * equal the registered entry's exactly (normally: both absent). Loopback
    40	 * ports stay port-agnostic (RFC 8252 §7.3) and one trailing slash is still
    41	 * ignored. The legacy fragment flow keeps the lenient matcher above.

    43	export function redirectUriMatchesAllowlistExact(
    44	    uri: string,
    45	    allowlist: readonly string[] | null | undefined,

    47	    if (!allowlist?.length) return false;
    48	    const incoming = safeParse(uri);
    49	    if (!incoming) return false;
    50	    return allowlist.some((entry) => matchesEntry(incoming, entry, true));
    51	}

   111	function matchesEntry(
   112	    incoming: URL,
   113	    entryUrl: string,
   114	    exactQuery = false,
   115	): boolean {
   116	    const entry = safeParse(entryUrl);
   117	    if (!entry) return false;
   118	    if (incoming.hash || entry.hash) return false;
   119	    if (incoming.protocol !== entry.protocol) return false;
   120	    if (
   121	        normalizeHostname(incoming.hostname) !==
   122	        normalizeHostname(entry.hostname)
   123	    ) {
   124	        return false;
   125	    }
   126	    if (
   127	        normalizePathname(incoming.pathname) !==
   128	        normalizePathname(entry.pathname)
   129	    ) {
   130	        return false;
   131	    }
   132	    if (exactQuery) {
   133	        if (incoming.search !== entry.search) return false;
   134	    } else if (entry.search && incoming.search !== entry.search) {
   135	        return false;
   136	    }
   137	    if (isLoopbackHostname(entry.hostname)) return true;
   138	    return incoming.port === entry.port;
   139	}

   107	export function isBlockedHostname(hostname: string): boolean {
   108	    const host = hostname
   109	        .replace(/^\[|\]$/g, "")
   110	        .replace(/\.$/, "")
   111	        .toLowerCase();
   112	    if (host === "localhost" || host.endsWith(".localhost")) return true;
   113	    if (host.endsWith(".local")) return true;
   114	    if (host.includes(":")) return true;
   115	    if (host.startsWith("127.") || host.startsWith("10.")) return true;
   116	    if (host.startsWith("169.254.")) return true;
   117	    if (host.startsWith("100.")) {
   118	        const second = Number(host.split(".")[1]);
   119	        if (second >= 64 && second <= 127) return true;
   120	    }
   121	    if (host.startsWith("192.168.")) return true;
   122	    const ipv4 = host.split(".").map(Number);
   123	    if (
   124	        ipv4.length === 4 &&
   125	        ipv4.every(
   126	            (part) => Number.isInteger(part) && part >= 0 && part <= 255,
   127	        ) &&
   128	        (ipv4[0] === 0 ||
   129	            (ipv4[0] ?? 0) >= 224 ||
   130	            (ipv4[0] === 198 && (ipv4[1] === 18 || ipv4[1] === 19)))
   131	    ) {
   132	        return true;
   133	    }
   134	    const match172 = host.match(/^172\.(\d+)\./);
   135	    if (match172) {
   136	        const second = Number(match172[1]);
   137	        if (second >= 16 && second <= 31) return true;
   138	    }
   139	    return false;

     1	const VERSION = "v1";
     2	
     3	export async function encryptSecret(
     4	    plaintext: string,
     5	    secret: string,
     6	): Promise<string> {
     7	    const iv = crypto.getRandomValues(new Uint8Array(12));
     8	    const key = await deriveKey(secret);
     9	    const ciphertext = new Uint8Array(
    10	        await crypto.subtle.encrypt(
    11	            { name: "AES-GCM", iv },
    12	            key,
    13	            new TextEncoder().encode(plaintext),
    14	        ),
    15	    );
    16	    return `${VERSION}:${bytesToBase64(iv)}:${bytesToBase64(ciphertext)}`;
    17	}
    18	
    19	export async function decryptSecret(
    20	    encrypted: string,
    21	    secret: string,
    22	): Promise<string> {
    23	    const [version, ivBase64, ciphertextBase64] = encrypted.split(":");
    24	    if (version !== VERSION || !ivBase64 || !ciphertextBase64) {
    25	        throw new Error("Unsupported encrypted secret format");
    26	    }
    27	    const key = await deriveKey(secret);
    28	    const plaintext = await crypto.subtle.decrypt(
    29	        { name: "AES-GCM", iv: base64ToBytes(ivBase64) },
    30	        key,
    31	        base64ToBytes(ciphertextBase64),
    32	    );
    33	    return new TextDecoder().decode(plaintext);
    34	}
    35	
    36	async function deriveKey(secret: string): Promise<CryptoKey> {
    37	    const digest = await crypto.subtle.digest(
    38	        "SHA-256",
    39	        new TextEncoder().encode(secret),
    40	    );
    41	    return crypto.subtle.importKey("raw", digest, "AES-GCM", false, [
    42	        "encrypt",
    43	        "decrypt",
    44	    ]);
    45	}
    46	
    47	function bytesToBase64(bytes: Uint8Array): string {
    48	    let binary = "";
    49	    for (const byte of bytes) binary += String.fromCharCode(byte);
    50	    return btoa(binary);
    51	}
    52	
    53	function base64ToBytes(value: string): Uint8Array<ArrayBuffer> {
    54	    const binary = atob(value);
    55	    const bytes = new Uint8Array(new ArrayBuffer(binary.length));
    56	    for (let i = 0; i < binary.length; i += 1) {
    57	        bytes[i] = binary.charCodeAt(i);
    58	    }
    59	    return bytes;
    60	}

    14	/**
    15	 * Atomically deducts pollen from user balance.
    16	 *
    17	 * This is the authoritative deduction ladder — the one place a positive charge
    18	 * picks a bucket and money actually moves, read top-to-bottom as the `decision`
    19	 * CASE below. `canCoverEstimatedCharge` answers the separate preflight question
    20	 * of whether an estimate is affordable, while `createBalanceCheckResult` only
    21	 * labels a meter for reporting. Neither is an authoritative write-path rule.
    22	 *
    23	 *   1. paid-only  → `pack` only (never Quest Pollen)
    24	 *   2. quest-only → `tier` only, for keys that never spend paid Pollen
    25	 *   3. restricted → `pack`, including debt for requests already admitted
    26	 *   4. tier       → Quest Pollen when it covers the full amount (`>=`)
    27	 *   5. pack       → any positive paid balance when Quest Pollen can't cover
    28	 *   6. tier       → fall back to Quest Pollen when pack is non-positive
    29	 *
    30	 * The final `tier` fall-through is load-bearing: when Quest Pollen cannot cover
    31	 * the charge and paid balance is non-positive, the charge accrues Quest-Pollen
    32	 * debt instead of reducing paid balance further. Non-positive amounts return
    33	 * without selecting a bucket or writing.
    34	 * Restrictions reject new Quest-only requests at preflight. Those already in
    35	 * flight still settle in their permitted bucket; completed work is never free.
    36	 */
    37	export async function atomicDeductUserBalance(
    38	    db: DrizzleD1Database,
    39	    userId: string,
    40	    amount: number,
    41	    isPaidOnly = false,
    42	    questPollenOnly = false,
    43	): Promise<{
    44	    ok: boolean;
    45	    bucket: Bucket | null;
    46	    packBalance: number | null;
    47	    lastPackPollen: number | null;
    48	    postDeductionBalance: number | null;
    49	}> {
    50	    if (amount <= 0) {
    51	        return {
    52	            ok: true,
    53	            bucket: null,
    54	            packBalance: null,
    55	            lastPackPollen: null,
    56	            postDeductionBalance: null,
    57	        };
    58	    }
    59	
    60	    const row = await db.get<{
    61	        bucket: Bucket;
    62	        tierBalance: number | null;
    63	        packBalance: number | null;
    64	        lastPackPollen: number | null;
    65	    }>(
    66	        sql`
    67	        WITH decision AS MATERIALIZED (
    68	            SELECT
    69	                id,
    70	                CASE
    71	                    WHEN ${isPaidOnly ? 1 : 0} = 1 THEN 'pack'
    72	                    WHEN ${questPollenOnly ? 1 : 0} = 1 THEN 'tier'
    73	                    WHEN quest_pollen_restricted = 1 THEN 'pack'
    74	                    WHEN COALESCE(tier_balance, 0) >= ${amount} THEN 'tier'
    75	                    WHEN COALESCE(pack_balance, 0) > 0 THEN 'pack'
    76	                    ELSE 'tier'
    77	                END AS bucket
    78	            FROM ${userTable}
    79	            WHERE id = ${userId}
    80	        )
    81	        UPDATE ${userTable}
    82	        SET
    83	            tier_balance = CASE
    84	                WHEN (SELECT bucket FROM decision) = 'tier'
    85	                    THEN ROUND(COALESCE(tier_balance, 0) - ${amount}, ${POLLEN_BILLING_PRECISION})
    86	                ELSE tier_balance
    87	            END,
    88	            pack_balance = CASE
    89	                WHEN (SELECT bucket FROM decision) = 'pack'
    90	                    THEN ROUND(COALESCE(pack_balance, 0) - ${amount}, ${POLLEN_BILLING_PRECISION})
    91	                ELSE pack_balance
    92	            END
    93	        WHERE id = (SELECT id FROM decision)
    94	        RETURNING
    95	            (SELECT bucket FROM decision) AS bucket,
    96	            tier_balance AS tierBalance,
    97	            pack_balance AS packBalance,
    98	            last_pack_pollen AS lastPackPollen
    99	    `,
   100	    );
   101	
   102	    return {
   103	        ok: !!row,
   104	        bucket: row?.bucket ?? null,
   105	        packBalance: row?.packBalance ?? null,
   106	        lastPackPollen: row?.lastPackPollen ?? null,
   107	        postDeductionBalance: row
   108	            ? row.bucket === "tier"
   109	                ? row.tierBalance
   110	                : row.packBalance
   111	            : null,
   112	    };
   113	}

     1	export type UserBalance = {
     2	    tierBalance: number;
     3	    packBalance: number;
     4	    questPollenRestricted?: boolean;
     5	};

     7	export type BalanceBucket = "tier" | "pack";

     9	/**
    10	 * Preflight "can the caller afford this estimate" check. This is intentionally
    11	 * a different question from the authoritative bucket ladder in
    12	 * `atomicDeductUserBalance`: that function chooses *which* bucket to write to
    13	 * when money actually moves; this one only asks whether *enough exists* to let
    14	 * the request proceed before any mutation.
    15	 *
    16	 * The two therefore need not use identical thresholds. This preflight clamps
    17	 * negative estimates to zero, but every request still requires a positive
    18	 * balance in at least one bucket. The write path skips non-positive charges
    19	 * entirely; for a positive actual charge, it uses
    20	 * Quest Pollen when that bucket covers the charge, otherwise any positive paid
    21	 * balance, and finally Quest Pollen debt when the paid balance is non-positive.
    22	 */
    23	export function canCoverEstimatedCharge(
    24	    balances: UserBalance,
    25	    estimatedCost: number,
    26	    isPaidOnly = false,
    27	): boolean {
    28	    const tierBalance = balances.questPollenRestricted
    29	        ? 0
    30	        : balances.tierBalance;
    31	    const hasPositiveBalance = tierBalance > 0 || balances.packBalance > 0;
    32	    if (!hasPositiveBalance) return false;
    33	
    34	    const threshold = Math.max(0, estimatedCost);
    35	    // Paid-only stays strict: the flag means "must hold paid balance", a
    36	    // positive-balance requirement rather than a cost-coverage one. A paid-only
    37	    // model is never free, so no zero-balance caller needs to get through.
    38	    if (isPaidOnly) return balances.packBalance > threshold;
    39	    // Once the wallet has a positive bucket, compare each bucket to the cost.
    40	    return tierBalance >= threshold || balances.packBalance >= threshold;
    41	}

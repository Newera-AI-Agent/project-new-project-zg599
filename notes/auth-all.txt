    81	/**
    82	 * Throws StagingAccessDeniedError if the env is staging and the user is not in
    83	 * either staging allowlist. No-op outside staging.
    84	 * Fails closed: a missing user or empty/missing allowlists deny access.

    86	 * Called at request-time (every API-key or session-cookie request) to defend
    87	 * against pre-existing sessions/keys that predate the lockdown. See #11137.

    89	export function assertStagingAccess(
    90	    env: {
    91	        ENVIRONMENT?: string;
    92	        STAGING_ALLOWED_GITHUB_IDS?: string;
    93	        STAGING_ALLOWED_EMAILS?: string;
    94	    },
    95	    user:
    96	        | { githubId?: number | null; email?: string | null }
    97	        | null
    98	        | undefined,
    99	): void {
   100	    if (env.ENVIRONMENT !== "staging") return;
   101	    const allowedGithubIds = parseGithubIdList(env.STAGING_ALLOWED_GITHUB_IDS);
   102	    const allowedEmails = parseEmailList(env.STAGING_ALLOWED_EMAILS);
   103	    const ghId = user?.githubId;
   104	    const email = normalizeEmail(user?.email);
   105	    if (
   106	        (!ghId || !allowedGithubIds.has(Number(ghId))) &&
   107	        (!email || !allowedEmails.has(email))
   108	    ) {
   109	        throw new StagingAccessDeniedError();
   110	    }
   111	}

   113	export function parseEmailList(raw: string | undefined | null): Set<string> {

   128	export function createApiKeyPlugin() {
   129	    return apiKey({
   130	        enableMetadata: true,
   131	        deferUpdates: true,
   132	        defaultPrefix: PUBLISHABLE_KEY_PREFIX,
   133	        defaultKeyLength: 16,
   134	        minimumNameLength: 1,
   135	        maximumNameLength: 253,
   136	        startingCharactersConfig: {
   137	            charactersLength: 10,
   138	        },
   139	        customKeyGenerator: (options: {
   140	            length: number;
   141	            prefix: string | undefined;
   142	        }) => {
   143	            const isPublishable = options.prefix === PUBLISHABLE_KEY_PREFIX;
   144	            const keyLength = isPublishable ? 16 : 32;
   145	            const chars =
   146	                "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789";
   147	            const randomBytes = crypto.getRandomValues(
   148	                new Uint8Array(keyLength),
   149	            );
   150	            const key = Array.from(
   151	                randomBytes,
   152	                (byte) => chars[byte % chars.length],
   153	            ).join("");
   154	            return options.prefix ? `${options.prefix}_${key}` : key;
   155	        },
   156	        keyExpiration: {
   157	            minExpiresIn: 0,
   158	            // Override Better Auth's 365-day default with JavaScript Date's range.
   159	            maxExpiresIn: 100_000_000,
   160	        },
   161	        rateLimit: {
   162	            enabled: false,
   163	        },
   164	    });
   165	}

   192	export function extractApiKey(request: Request): string | null {
   193	    const auth = request.headers.get("authorization");
   194	    const match = auth?.match(/^Bearer (.+)$/);
   195	    if (match?.[1]) return match[1];
   196	
   197	    // E2B's SDKs send the key in X-API-KEY.
   198	    const headerKey = request.headers.get("x-api-key");
   199	    if (headerKey) return headerKey;
   200	
   201	    // Query keys end up in access logs, referrers and browser history. Their
   202	    // owner can rotate them; agent run and session tokens cannot be revoked,
   203	    // so they are Bearer-only.
   204	    const queryKey = new URL(request.url).searchParams.get("key");
   205	    return queryKey?.startsWith(AGENT_RUN_TOKEN_PREFIX) ||
   206	        queryKey?.startsWith(SESSION_TOKEN_PREFIX)
   207	        ? null
   208	        : queryKey;
   209	}

   229	export async function authenticateApiKeyRequest(opts: {
   230	    request: Request;
   231	    env: ApiKeyAuthBindings;
   232	    ctx?: Pick<ExecutionContext, "waitUntil">;
   233	}): Promise<ApiKeyAuthResult | null> {
   234	    const rawApiKey = extractApiKey(opts.request);
   235	    if (!rawApiKey) return null;
   236	
   237	    // pk_ keys are App Keys: public OAuth client IDs, never credentials.
   238	    if (rawApiKey.startsWith(`${PUBLISHABLE_KEY_PREFIX}_`)) {
   239	        throw new HTTPException(401, { message: PUBLISHABLE_KEY_REFUSAL });
   240	    }
   241	
   242	    if (rawApiKey.startsWith(AGENT_RUN_TOKEN_PREFIX)) {
   243	        return authenticateAgentRunToken(rawApiKey, opts.env);
   244	    }
   245	    if (rawApiKey.startsWith(SESSION_TOKEN_PREFIX)) {
   246	        return authenticateSessionToken(rawApiKey, opts.env);
   247	    }
   248	
   249	    // Better Auth's verifyApiKey does this lookup plus two awaited writes per
   250	    // call (last_request and updated_at). Our keys use none of its quota or
   251	    // rate-limit features, so look the key up by hash directly.
   252	    const keyHash = await defaultKeyHasher(rawApiKey);
   253	    const invalidKeyCacheKey = new Request(
   254	        new URL(`/__invalid-api-key/${keyHash}`, opts.request.url),
   255	    );
   256	    const invalidKeys = await caches.open("invalid-api-keys");
   257	    if (await invalidKeys.match(invalidKeyCacheKey)) return null;
   258	
   259	    const row = await selectApiKeyRow(opts.env, eq(schema.apikey.key, keyHash));
   260	    const result = toActiveAuthResult(row, rawApiKey, opts.env);
   261	    if (!result || !row) {
   262	        const remember = invalidKeys.put(
   263	            invalidKeyCacheKey,
   264	            new Response(null, {
   265	                headers: {
   266	                    "Cache-Control": `max-age=${INVALID_KEY_CACHE_SECONDS}`,
   267	                },
   268	            }),
   269	        );
   270	        runInBackground(remember, opts.ctx);
   271	        return null;
   272	    }
   273	
   274	    const lastRequest = row.apiKey.lastRequest;
   275	    const staleBefore = new Date(Date.now() - LAST_REQUEST_WRITE_INTERVAL_MS);
   276	    if (!lastRequest || lastRequest < staleBefore) {
   277	        const touch = drizzle(opts.env.DB)
   278	            .update(schema.apikey)
   279	            .set({ lastRequest: new Date() })
   280	            .where(
   281	                and(
   282	                    eq(schema.apikey.id, row.apiKey.id),
   283	                    or(
   284	                        isNull(schema.apikey.lastRequest),
   285	                        lt(schema.apikey.lastRequest, staleBefore),
   286	                    ),
   287	                ),
   288	            )
   289	            .run();
   290	        runInBackground(touch, opts.ctx);
   291	    }
   292	    return result;
   293	}

   327	async function authenticateAgentRunToken(
   328	    rawToken: string,
   329	    env: ApiKeyAuthBindings,
   330	): Promise<ApiKeyAuthResult | null> {
   331	    if (!env.BETTER_AUTH_SECRET) return null;
   332	
   333	    let claims: AgentRunClaims;
   334	    try {
   335	        claims = await verifyAgentRunToken(rawToken, env.BETTER_AUTH_SECRET);
   336	    } catch {
   337	        return null;
   338	    }
   339	
   340	    const parent = await loadActiveApiKeyAuthResult({
   341	        apiKeyId: claims.parentApiKeyId,
   342	        rawApiKey: rawToken,
   343	        env,
   344	    });
   345	    if (!parent) return null;
   346	
   347	    // Agent-run tokens leave our control. Keep the parent's generation limits,
   348	    // but never let an agent manage the owner's account or keys.
   349	    const models = parent.apiKey.permissions?.models;
   350	    return {
   351	        ...parent,
   352	        apiKey: {
   353	            ...parent.apiKey,
   354	            permissions: models ? { models } : undefined,
   355	        },
   356	        agentRun: claims,
   357	    };
   358	}

   404	async function selectApiKeyRow(env: ApiKeyAuthBindings, where: SQL) {
   405	    const db = drizzle(env.DB, { schema });
   406	    const byopClientKey = alias(schema.apikey, "byop_client_key");
   407	    const byopOwner = alias(schema.user, "byop_owner");
   408	    return db
   409	        .select({
   410	            apiKey: getTableColumns(schema.apikey),
   411	            user: getTableColumns(schema.user),
   412	            byopClientName: byopClientKey.name,
   413	            byopOwner: {
   414	                banned: byopOwner.banned,
   415	                banExpires: byopOwner.banExpires,
   416	            },
   417	            byopClientUserId: byopClientKey.referenceId,
   418	            byopClientPrefix: byopClientKey.prefix,
   419	            byopClientEnabled: byopClientKey.enabled,
   420	            byopClientExpiresAt: byopClientKey.expiresAt,
   421	            byopClientMetadata: byopClientKey.metadata,
   422	        })
   423	        .from(schema.apikey)
   424	        .innerJoin(schema.user, eq(schema.user.id, schema.apikey.referenceId))
   425	        .leftJoin(
   426	            byopClientKey,
   427	            eq(byopClientKey.id, schema.apikey.byopClientKeyId),
   428	        )
   429	        .where(where)
   430	        .leftJoin(byopOwner, eq(byopOwner.id, byopClientKey.referenceId))
   431	        .get();
   432	}

   434	function toActiveAuthResult(
   435	    row: ApiKeyRow,
   436	    rawApiKey: string,
   437	    env: ApiKeyAuthBindings,
   438	): (ApiKeyAuthResult & { apiKey: AuthenticatedApiKey }) | null {
   439	    if (!row || !isUsable(row.apiKey)) return null;
   440	    // The stored type, not the prefix: older App Keys look like plln_pk_….
   441	    const metadata = normalizeMetadata(parseMetadata(row.apiKey.metadata));
   442	    if (metadata?.keyType === "publishable") {
   443	        throw new HTTPException(401, { message: PUBLISHABLE_KEY_REFUSAL });
   444	    }
   445	
   446	    assertNotBanned(row.user);
   447	    if (row.byopOwner) assertNotBanned(row.byopOwner);
   448	    assertStagingAccess(env, row.user);
   449	
   450	    return {
   451	        user: row.user,
   452	        apiKey: {
   453	            id: row.apiKey.id,
   454	            name: row.apiKey.name ?? undefined,
   455	            permissions: normalizePermissions(
   456	                parseMetadata(row.apiKey.permissions),
   457	            ),
   458	            metadata,
   459	            pollenBalance: row.apiKey.pollenBalance ?? null,
   460	            expiresAt: row.apiKey.expiresAt,
   461	            questPollenOnly: row.apiKey.questPollenOnly,
   462	            byopClientKeyId: row.apiKey.byopClientKeyId ?? null,
   463	            byopClientName: row.byopClientName ?? null,
   464	            byopClientUserId: row.byopClientUserId ?? null,
   465	            byopMarkupApplies: byopClientAllowsMarkup(
   466	                {
   467	                    userId: row.byopClientUserId,
   468	                    prefix: row.byopClientPrefix,
   469	                    enabled: row.byopClientEnabled,
   470	                    expiresAt: row.byopClientExpiresAt,
   471	                    metadata: row.byopClientMetadata,
   472	                },
   473	                row.user.id,
   474	            ),
   475	            rawKey: rawApiKey,
   476	        },
   477	        rawApiKey,
   478	    };
   479	}

   481	function normalizePermissions(
   482	    value: unknown,
   483	): Record<string, string[]> | undefined {
   484	    if (!value || typeof value !== "object" || Array.isArray(value)) {
   485	        return undefined;
   486	    }
   487	
   488	    const permissions: Record<string, string[]> = {};
   489	    for (const [key, scopes] of Object.entries(value)) {
   490	        if (!Array.isArray(scopes)) continue;
   491	        const safeScopes = scopes.filter(
   492	            (scope): scope is string => typeof scope === "string",
   493	        );
   494	        if (safeScopes.length || key === "models") {
   495	            permissions[key] = safeScopes;
   496	        }
   497	    }
   498	    return Object.keys(permissions).length ? permissions : undefined;
   499	}

   505	        return undefined;
   506	    }
   507	    return value as Record<string, unknown>;

    72	export function validateRedirectUriFormat(redirectUri: string): void {
    73	    if (!/^[a-z][a-z0-9+\-.]*:\/\/.+/.test(redirectUri)) {
    74	        throw new HTTPException(400, {
    75	            message: "Must be a valid URL with a scheme (e.g. https://...)",
    76	        });
    77	    }
    78	    let parsed: URL;
    79	    try {
    80	        parsed = new URL(redirectUri);
    81	    } catch {
    82	        throw new HTTPException(400, {
    83	            message: "Must be a valid URL with a scheme (e.g. https://...)",
    84	        });
    85	    }
    86	    if (parsed.hash) {
    87	        throw new HTTPException(400, {

    98	        throw new HTTPException(400, {
    99	            message:
   100	                "Redirect URI must use https://, except http:// is allowed for loopback hosts",
   101	        });
   102	    }
   103	}

   171	async function validateClientRedirectBinding(
   172	    authClient: CreateApiKeyAuthClient,
   173	    db: ApiKeyDb,
   174	    metadata: CallerMetadata | undefined,
   175	): Promise<VerifiedClientAttribution | null> {
   176	    if (!metadata) return null;
   177	    const requestedClientId = metadata.requestedClientId;
   178	    if (typeof requestedClientId !== "string") {
   179	        return null;
   180	    }
   181	
   182	    const clientKey = await findPublishableClientKey(
   183	        authClient,
   184	        db,
   185	        requestedClientId,
   186	    );
   187	    if (!clientKey) rejectInvalidClientId();
   188	    const attribution = {
   189	        clientId: clientKey.id,
   190	    };
   191	
   192	    if (typeof metadata.deviceUserCode === "string") {
   193	        const device = await db.query.deviceCode.findFirst({
   194	            where: eq(
   195	                schema.deviceCode.userCode,
   196	                metadata.deviceUserCode.toUpperCase(),
   197	            ),
   198	        });
   199	        if (
   200	            !device ||
   201	            device.status !== "pending" ||
   202	            device.expiresAt < new Date() ||
   203	            device.clientId !== requestedClientId
   204	        ) {
   205	            throw new HTTPException(400, {
   206	                message: "client_id mismatch",
   207	            });
   208	        }
   209	        return attribution;
   210	    }
   211	
   212	    if (typeof metadata.redirectUri !== "string") {
   213	        throw new HTTPException(400, {
   214	            message: "redirect_uri is required when client_id is provided",
   215	        });
   216	    }
   217	    // No scheme/format check on the incoming redirect_uri here: scheme policy is
   218	    // enforced at registration (validateRedirectUriFormat on the app key's
   219	    // redirectUris). The allowlist match below is the only gate the auth flow
   220	    // applies, so a registered URI of any kind round-trips unchanged.
   221	    const allowlist = getRedirectUris(parseMetadata(clientKey.metadata));
   222	    if (!redirectUriMatchesAllowlist(metadata.redirectUri, allowlist)) {
   223	        throw new HTTPException(400, {
   224	            message: "redirect_uri not in allowlist for this client_id",
   225	        });
   226	    }
   227	    return attribution;
   228	}

   231	    authClient,
   232	    dbBinding,
   233	    userId,
   234	    name,
   235	    type,
   236	    expiresIn,
   237	    allowedModels,

===== shared/auth/redirect-uri.ts =====
     1	/**
     2	 * Match an incoming `redirect_uri` against a registered allowlist for a `pk_`.
     3	 *
     4	 * Scheme policy (https-only except loopback http) is enforced at registration
     5	 * time (see validateRedirectUriFormat). Matching trusts the stored allowlist
     6	 * and does not re-check the scheme.
     7	 *
     8	 * Comparison rules:
     9	 * - Scheme + hostname + path are matched exactly (after lowercasing host).
    10	 * - Trailing slash on the path is insignificant: `/cb` and `/cb/` match.
    11	 * - Query strings are ignored when the registered URI has no query: apps
    12	 *   round-trip state (e.g. `?prompt=...`) and the auth code/token rides the URL
    13	 *   fragment, not the query. If the registered URI includes a query, it must
    14	 *   match exactly so existing query-bound allowlist entries are not broadened.
    15	 * - For loopback entries (RFC 8252 §7.3): port is ignored — any port matches.
    16	 *   This covers native/CLI apps that bind to ephemeral ports each run.
    17	 * - For non-loopback entries: port must match (default ports normalized).
    18	 * - Fragments are rejected; redirect URIs must not contain them.
    19	 *
    20	 * Returns false on any parse error or empty allowlist.
    21	 *
    22	 * Single source of truth for both /api/app-lookup (consent screen attribution)
    23	 * and createApiKeyForUser (sk_ minting). Keep them aligned.
    24	 */
    25	export function redirectUriMatchesAllowlist(
    26	    uri: string,
    27	    allowlist: readonly string[] | null | undefined,
    28	): boolean {
    29	    if (!allowlist?.length) return false;
    30	    const incoming = safeParse(uri);
    31	    if (!incoming) return false;
    32	    return allowlist.some((entry) => matchesEntry(incoming, entry));
    33	}
    34	
    35	/**
    36	 * Strict variant for the OAuth authorization-code flow (RFC 6749 §3.1.2.3):
    37	 * the authorization code rides the redirect query string, so the fragment-era
    38	 * leniency of accepting extra query params must not apply. The query must
    39	 * equal the registered entry's exactly (normally: both absent). Loopback
    40	 * ports stay port-agnostic (RFC 8252 §7.3) and one trailing slash is still
    41	 * ignored. The legacy fragment flow keeps the lenient matcher above.
    42	 */
    43	export function redirectUriMatchesAllowlistExact(
    44	    uri: string,
    45	    allowlist: readonly string[] | null | undefined,
    46	): boolean {
    47	    if (!allowlist?.length) return false;
    48	    const incoming = safeParse(uri);
    49	    if (!incoming) return false;
    50	    return allowlist.some((entry) => matchesEntry(incoming, entry, true));
    51	}

    65	/**
    66	 * Parse the redirect target of the consent screen: a website, a loopback
    67	 * server or a native app's own scheme (e.g. `myapp://callback`). Null when the
    68	 * URL is unparseable or uses a scheme the browser handles itself, so such a

    78	export function isAllowedRedirectUrl(url: URL): boolean {
    79	    if (url.protocol === "https:") return true;
    80	    return url.protocol === "http:" && isLoopbackHostname(url.hostname);

    83	export function isLoopbackHostname(hostname: string): boolean {
    84	    const h = normalizeHostname(hostname);
    85	    if (h === "localhost" || h === "0.0.0.0" || h === "::1") return true;
    86	    if (/^127\.\d+\.\d+\.\d+$/.test(h)) return true;
    87	    return false;
    88	}

    90	function safeParse(url: string): URL | null {
    91	    try {
    92	        return new URL(url);
    93	    } catch {
    94	        return null;
    95	    }
    96	}

    98	function normalizeHostname(hostname: string): string {
    99	    return hostname
   100	        .toLowerCase()
   101	        .replace(/^\[(.*)\]$/, "$1")
   102	        .replace(/\.$/, "");
   103	}
   104	
   105	function normalizePathname(pathname: string): string {
   106	    return pathname.length > 1 && pathname.endsWith("/")
   107	        ? pathname.slice(0, -1)
   108	        : pathname;
   109	}
   110	
   111	function matchesEntry(
   112	    incoming: URL,
   113	    entryUrl: string,
   114	    exactQuery = false,
   115	): boolean {
   116	    const entry = safeParse(entryUrl);
   117	    if (!entry) return false;
   118	    if (incoming.hash || entry.hash) return false;
   119	    if (incoming.protocol !== entry.protocol) return false;
   120	    if (
   121	        normalizeHostname(incoming.hostname) !==
   122	        normalizeHostname(entry.hostname)
   123	    ) {
   124	        return false;
   125	    }
   126	    if (
   127	        normalizePathname(incoming.pathname) !==
   128	        normalizePathname(entry.pathname)
   129	    ) {
   130	        return false;
   131	    }
   132	    if (exactQuery) {
   133	        if (incoming.search !== entry.search) return false;
   134	    } else if (entry.search && incoming.search !== entry.search) {
   135	        return false;
   136	    }
   137	    if (isLoopbackHostname(entry.hostname)) return true;
   138	    return incoming.port === entry.port;
   139	}

   130	 * Signs that a run of this agent was refused this grant on this key. The grant
   131	 * link reaches the owner through the agent, so an unsigned name would let one
   132	 * agent pose as another. Binding the key and the grant stops a signed name
   133	 * from being moved onto a different request.
   134	 */
   135	export function signGrantAgent(opts: {
   136	    secret: string;
   137	    apiKeyId: string;
   138	    grant: Grant;
   139	    agent: string;
   140	}): string {
   141	    const { grant } = opts;
   142	    const entry =
   143	        "category" in grant
   144	            ? `category:${grant.category}`
   145	            : "model" in grant
   146	              ? `model:${grant.model}`
   147	              : `permission:${grant.permission}`;
   148	    return createHmac("sha256", `pollinations-grant-agent:v1\0${opts.secret}`)
   149	        .update([opts.apiKeyId, entry, opts.agent].join("\0"))
   150	        .digest("base64url");
   151	}
   152	
   153	export function verifyGrantAgent(
   154	    opts: Parameters<typeof signGrantAgent>[0] & { sig: string },
   155	): boolean {
   156	    const expected = Buffer.from(signGrantAgent(opts));
   157	    const actual = Buffer.from(opts.sig);
   158	    return (
   159	        expected.length === actual.length && timingSafeEqual(expected, actual)
   160	    );
   161	}

    14	export function hasAccountPermission(
    15	    apiKey: AccountPermissionApiKey | undefined,
    16	    permission: AccountPermission,
    17	): boolean {
    18	    if (!apiKey) return true;
    19	    return !!apiKey.permissions?.account?.includes(permission);
    20	}
    21	
    22	/** Refuses a key without the permission, with a link to grant it to that key. */
    23	export function requireAccountPermission(
    24	    c: RefusalContext,
    25	    permission: AccountPermission,
    26	): void {
    27	    if (hasAccountPermission(c.var.auth?.apiKey, permission)) return;
    28	    // Agent run tokens never carry account permissions (api-key.ts), so a
    29	    // grant would only widen the owner's key without letting the agent in.
    30	    if (c.var.auth?.agentRun) {
    31	        throw new HTTPException(403, {
    32	            message: `Agent runs can't use 'account:${permission}'.`,
    33	        });
    34	    }
    35	    throw permissionRequired(
    36	        c,
    37	        { permission },
    38	        `API key does not have 'account:${permission}' permission.`,
    39	    );
    40	}

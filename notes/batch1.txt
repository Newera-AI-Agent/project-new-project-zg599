    14	/**
    15	 * Cloudflare-native limit that runs before auth, so one flooding key is
    16	 * rejected without reaching D1. Counts per API key, hashed so no lookup is
    17	 * needed. With `byIp`, requests without a key count per client IP instead,
    18	 * and keyed requests also count per IP in `EDGE_IP_RATE_LIMITER`, so
    19	 * rotating made-up keys from one IP cannot dodge the limit. Each Worker's
    20	 * wrangler.toml sets its limits per 10 seconds, matching `Retry-After`.
    21	 * Enter leaves `byIp` off because gen's internal calls arrive without a
    22	 * client IP; it still counts every request that carries a client IP, keyed
    23	 * or not, so public routes such as device codes, app lookup and sign-in
    24	 * cannot be flooded into D1 with made-up keys. Cloudflare counts per
    25	 * location, so this caps floods rather than enforcing an exact quota.
    26	 */
    27	export function rateLimit({ byIp }: { byIp: boolean }) {
    28	    return createMiddleware<RateLimitEnv>(async (c, next) => {
    29	        const apiKey = extractApiKey(c.req.raw);
    30	        const ip = `ip:${getRealClientIp(c)}`;
    31	        if (apiKey) {
    32	            const digest = await crypto.subtle.digest(
    33	                "SHA-256",
    34	                new TextEncoder().encode(apiKey),
    35	            );
    36	            await limit(
    37	                c,
    38	                c.env.EDGE_RATE_LIMITER,
    39	                `key:${bytesToHex(digest)}`,
    40	            );
    41	            if (byIp) await limit(c, c.env.EDGE_IP_RATE_LIMITER, ip);
    42	        } else if (byIp) {
    43	            await limit(c, c.env.EDGE_RATE_LIMITER, ip);
    44	        }
    45	        if (!byIp && c.req.header("cf-connecting-ip")) {
    46	            await limit(c, c.env.EDGE_RATE_LIMITER, ip);
    47	        }
    48	        await next();
    49	    });
    50	}
    51	
    52	/** A 429 once `key` is over `limiter`; `retryAfter` is its period in seconds. */
    53	export async function limit(
    54	    c: Context,
    55	    limiter: RateLimit | undefined,
    56	    key: string,
    57	    retryAfter = 10,
    58	) {
    59	    if (limiter && !(await limiter.limit({ key })).success) {
    60	        c.header("Retry-After", String(retryAfter));

    44	        }

     1	/**
     2	 * Match an incoming `redirect_uri` against a registered allowlist for a `pk_`.
     3	 *
     4	 * Scheme policy (https-only except loopback http) is enforced at registration
     5	 * time (see validateRedirectUriFormat). Matching trusts the stored allowlist
     6	 * and does not re-check the scheme.
     7	 *
     8	 * Comparison rules:
     9	 * - Scheme + hostname + path are matched exactly (after lowercasing host).
    10	 * - Trailing slash on the path is insignificant: `/cb` and `/cb/` match.
    11	 * - Query strings are ignored when the registered URI has no query: apps
    12	 *   round-trip state (e.g. `?prompt=...`) and the auth code/token rides the URL
    13	 *   fragment, not the query. If the registered URI includes a query, it must
    14	 *   match exactly so existing query-bound allowlist entries are not broadened.
    15	 * - For loopback entries (RFC 8252 §7.3): port is ignored — any port matches.
    16	 *   This covers native/CLI apps that bind to ephemeral ports each run.
    17	 * - For non-loopback entries: port must match (default ports normalized).
    18	 * - Fragments are rejected; redirect URIs must not contain them.
    19	 *
    20	 * Returns false on any parse error or empty allowlist.
    21	 *
    22	 * Single source of truth for both /api/app-lookup (consent screen attribution)
    23	 * and createApiKeyForUser (sk_ minting). Keep them aligned.
    24	 */
    25	export function redirectUriMatchesAllowlist(
    26	    uri: string,
    27	    allowlist: readonly string[] | null | undefined,
    28	): boolean {
    29	    if (!allowlist?.length) return false;
    30	    const incoming = safeParse(uri);
    31	    if (!incoming) return false;
    32	    return allowlist.some((entry) => matchesEntry(incoming, entry));
    33	}

   111	function matchesEntry(
   112	    incoming: URL,
   113	    entryUrl: string,
   114	    exactQuery = false,
   115	): boolean {
   116	    const entry = safeParse(entryUrl);
   117	    if (!entry) return false;
   118	    if (incoming.hash || entry.hash) return false;
   119	    if (incoming.protocol !== entry.protocol) return false;
   120	    if (
   121	        normalizeHostname(incoming.hostname) !==
   122	        normalizeHostname(entry.hostname)
   123	    ) {
   124	        return false;
   125	    }
   126	    if (
   127	        normalizePathname(incoming.pathname) !==
   128	        normalizePathname(entry.pathname)
   129	    ) {
   130	        return false;
   131	    }
   132	    if (exactQuery) {
   133	        if (incoming.search !== entry.search) return false;
   134	    } else if (entry.search && incoming.search !== entry.search) {
   135	        return false;
   136	    }
   137	    if (isLoopbackHostname(entry.hostname)) return true;
   138	    return incoming.port === entry.port;
   139	}

     3	export async function encryptSecret(
     4	    plaintext: string,
     5	    secret: string,
     6	): Promise<string> {
     7	    const iv = crypto.getRandomValues(new Uint8Array(12));

     9	    const ciphertext = new Uint8Array(
    10	        await crypto.subtle.encrypt(

    12	            key,

    14	        ),
    15	    );

    17	}

    19	export async function decryptSecret(
    20	    encrypted: string,
    21	    secret: string,
    22	): Promise<string> {
    23	    const [version, ivBase64, ciphertextBase64] = encrypted.split(":");
    24	    if (version !== VERSION || !ivBase64 || !ciphertextBase64) {
    25	        throw new Error("Unsupported encrypted secret format");
    26	    }
    27	    const key = await deriveKey(secret);
    28	    const plaintext = await crypto.subtle.decrypt(
    29	        { name: "AES-GCM", iv: base64ToBytes(ivBase64) },
    30	        key,
    31	        base64ToBytes(ciphertextBase64),
    32	    );
    33	    return new TextDecoder().decode(plaintext);
    34	}

    42	        "encrypt",
    43	        "decrypt",

   323	let methodRouter: PatternRouter<string> | undefined;

   325	/** The methods some route serves at this path. */
   326	function allowedMethods(path: string): Set<string> {
   327	    if (!methodRouter) {
   328	        methodRouter = new PatternRouter<string>();
   329	        for (const route of app.routes) {
   330	            if (route.method !== "ALL") {
   331	                methodRouter.add("ALL", route.path, route.method);
   332	            }
   333	        }
   334	    }
   335	    const methods = new Set(
   336	        methodRouter.match("GET", path)[0].map(([method]) => method),
   337	    );
   338	    if (methods.has("GET")) methods.add("HEAD");
   339	    return methods;
   340	}

   342	app.notFound(async (c: Context<Env>) => {
   343	    // Gen's icons and images in public/.
   344	    if (c.req.method === "GET" || c.req.method === "HEAD") {
   345	        const asset = await c.env.ASSETS.fetch(c.req.raw);
   346	        if (asset.status !== 404) return asset;
   347	    }
   348	    // A known path called with the wrong method is a 405, not a 404
   349	    // (RFC 9110 §15.5.6), so a GET probe of POST /v1/chat/completions
   350	    // does not conclude the endpoint is missing.
   351	    const allow = allowedMethods(c.req.path);
   352	    if (allow.size > 0 && !allow.has(c.req.method)) {
   353	        const response = await handleError(new HTTPException(405), c);
   354	        response.headers.set("Allow", [...allow].join(", "));
   355	        return response;
   356	    }
   357	    return handleError(new HTTPException(404), c);
   358	});

   363	    fetch: (request: Request, env, ctx) =>
   364	        withD1Calls(
   365	            env,
   366	            ctx,
   367	            () => ({ route_path: "", method: request.method }),
   368	            () => {
   369	                // A port host belongs to a sandbox, a site's host to the site.
   370	                const port = sandboxPortHost(
   371	                    env,
   372	                    new URL(request.url).hostname,

   374	                if (port) {
   375	                    return new Hono<Env>()
   376	                        .use("*", edgeRateLimit)
   377	                        .all("*", () => servePort(request, env, ctx, port))
   378	                        .onError(handleError)
   379	                        .fetch(request, env, ctx);
   380	                }
   381	                return serveSite(request, env) ?? app.fetch(request, env, ctx);
   382	            },
   383	        ),
   384	    scheduled: (controller, env, ctx) =>
   385	        withD1Calls(
   386	            env,
   387	            ctx,
   388	            () => ({ route_path: controller.cron, method: "CRON" }),
   389	            () => runScheduled(controller, env, ctx),

   393	async function runScheduled(
   394	    controller: ScheduledController,
   395	    env: CloudflareBindings,
   396	    ctx: ExecutionContext,
   397	): Promise<void> {
   398	    if (controller.cron === "* * * * *") {
   399	        // infra-watch reads the heartbeat's age to spot a failing refresh.
   400	        await refreshCommunityModelRegistryCache(env)
   401	            .then(() =>
   402	                env.KV.put(
   403	                    "heartbeat:community-catalog",
   404	                    Date.now().toString(),
   405	                ),
   406	            )
   407	            .catch((error) => {
   408	                console.error("Community model catalog refresh failed", error);
   409	            });
   410	        return;
   411	    }
   412	    // Every 5 minutes. Leases end only after the keeper has renewed the
   413	    // kept sandboxes; catalog refreshes do not change their cadence. A
   414	    // failed keeper still stops expired boxes: a lease outlives one run.
   415	    try {
   416	        await keepSandboxes(env, ctx);
   417	    } finally {
   418	        await stopExpiredAzureSandboxes(env);
   419	    }
   420	    await env.KV.put("heartbeat:sandbox-keeper", Date.now().toString());
   421	}
